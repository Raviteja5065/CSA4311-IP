PortalServlet.java import java.io.*; import javax.servlet.*; import javax.servlet.http.*; import java.util.*; public class PortalServlet extends HttpServlet { ArrayList jobs = new ArrayList<>(); ArrayList applicants = new ArrayList<>(); protected void doPost(HttpServletRequest req, HttpServletResponse res) throws IOException { res.setContentType("text/html"); PrintWriter out = res.getWriter(); String action = req.getParameter("action"); String name = req.getParameter("name"); HttpSession session = req.getSession(); if ("login".equals(action)) { session.setAttribute("user", name); out.println("Login Successful"); } else if ("register".equals(action)) { out.println("Registration Successful"); } else if ("post".equals(action)) { jobs.add(req.getParameter("vacancy")); out.println("Vacancy Posted Successfully"); } else if ("search".equals(action)) { for (String job : jobs) out.println(job + "
"); } else if ("apply".equals(action)) { applicants.add((String)session.getAttribute("user")); out.println("Application Submitted"); } else if ("view".equals(action)) { for (String a : applicants) out.println(a + "
"); } else if ("approve".equals(action)) { out.println("Director Approved"); } else if ("logout".equals(action)) { session.invalidate(); out.println("Logged Out Successfully"); } } } 4. WEB-INF/web.xml PortalServlet PortalServlet PortalServlet /portal
